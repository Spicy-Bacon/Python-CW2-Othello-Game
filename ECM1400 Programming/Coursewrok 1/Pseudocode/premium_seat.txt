function print_plane(plane)

    set row_letters [a b c d e f g h i j k l m n o p q r]

    print 1 2 3   4 5 6 #print the seat number on the top of the representation

    for row_no from 1 to len(plane)
        set seats = plane[row_no - 1]
        set row_letter = row_letters[row_no - 1] 

        print row_letter + ' ' #print the row letter on the left of the representation

        for seat_no from 1 to 6
            print seats[seat_no - 1]

            if seat_no == 3 then
                print '  |  '
            else
                print ' ' 
            end if
        end for

        print new line

        if row_no == 10 then
            print 'emergency exit'
        end if
    end for

end function
